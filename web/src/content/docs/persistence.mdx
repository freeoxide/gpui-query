# Persistence

gpui-query supports persistent caching through the `QueryPersister` trait. Implement this trait to store query data in any backend: files, databases, browser storage, or custom solutions.

## QueryPersister Trait

```rust
pub trait QueryPersister: Send + Sync + 'static {
    fn persist(
        &self,
        key: &QueryKey,
        data: &[u8],
    ) -> impl Future<Output = Result<(), PersistenceError>> + Send;

    fn restore(
        &self,
        key: &QueryKey,
    ) -> impl Future<Output = Result<Option<Vec<u8>>, PersistenceError>> + Send;

    fn remove(
        &self,
        key: &QueryKey,
    ) -> impl Future<Output = Result<(), PersistenceError>> + Send;

    fn clear(&self) -> impl Future<Output = Result<(), PersistenceError>> + Send;
}
```

## Implementing a Custom Persister

### File-Based Persister

Store cached data as files on disk:

```rust
use std::path::PathBuf;
use gpui_query::persistence::{QueryPersister, PersistenceError};

struct FilePersister {
    base_dir: PathBuf,
}

impl FilePersister {
    fn new(base_dir: impl Into<PathBuf>) -> Self {
        Self { base_dir: base_dir.into() }
    }

    fn key_to_path(&self, key: &QueryKey) -> PathBuf {
        let hash = key.hash_hex();
        self.base_dir.join(format!("{}.cache", hash))
    }
}

impl QueryPersister for FilePersister {
    async fn persist(
        &self,
        key: &QueryKey,
        data: &[u8],
    ) -> Result<(), PersistenceError> {
        let path = self.key_to_path(key);
        tokio::fs::write(&path, data).await
            .map_err(|e| PersistenceError::WriteFailed(e.to_string()))?;
        Ok(())
    }

    async fn restore(
        &self,
        key: &QueryKey,
    ) -> Result<Option<Vec<u8>>, PersistenceError> {
        let path = self.key_to_path(key);
        match tokio::fs::read(&path).await {
            Ok(data) => Ok(Some(data)),
            Err(_) => Ok(None),
        }
    }

    async fn remove(
        &self,
        key: &QueryKey,
    ) -> Result<(), PersistenceError> {
        let path = self.key_to_path(key);
        tokio::fs::remove_file(&path).await
            .map_err(|e| PersistenceError::RemoveFailed(e.to_string()))?;
        Ok(())
    }

    async fn clear(&self) -> Result<(), PersistenceError> {
        let mut entries = tokio::fs::read_dir(&self.base_dir).await
            .map_err(|e| PersistenceError::ClearFailed(e.to_string()))?;
        while let Some(entry) = entries.next_entry().await
            .map_err(|e| PersistenceError::ClearFailed(e.to_string()))?
        {
            if entry.path().extension().map_or(false, |ext| ext == "cache") {
                tokio::fs::remove_file(entry.path()).await.ok();
            }
        }
        Ok(())
    }
}
```

### Database Persister

Store cached data in SQLite:

```rust
struct SqlitePersister {
    pool: SqlitePool,
}

impl SqlitePersister {
    async fn new(database_url: &str) -> Result<Self, sqlx::Error> {
        let pool = SqlitePool::connect(database_url).await?;
        sqlx::query(
            "CREATE TABLE IF NOT EXISTS query_cache (
                key_hash TEXT PRIMARY KEY,
                data BLOB NOT NULL,
                created_at INTEGER NOT NULL
            )"
        )
        .execute(&pool)
        .await?;
        Ok(Self { pool })
    }
}

impl QueryPersister for SqlitePersister {
    async fn persist(&self, key: &QueryKey, data: &[u8]) -> Result<(), PersistenceError> {
        let hash = key.hash_hex();
        sqlx::query(
            "INSERT OR REPLACE INTO query_cache (key_hash, data, created_at) VALUES (?, ?, ?)"
        )
        .bind(&hash)
        .bind(data)
        .bind(chrono::Utc::now().timestamp())
        .execute(&self.pool)
        .await
        .map_err(|e| PersistenceError::WriteFailed(e.to_string()))?;
        Ok(())
    }

    async fn restore(&self, key: &QueryKey) -> Result<Option<Vec<u8>>, PersistenceError> {
        let hash = key.hash_hex();
        let row = sqlx::query_as::<_, (Vec<u8>,)>(
            "SELECT data FROM query_cache WHERE key_hash = ?"
        )
        .bind(&hash)
        .fetch_optional(&self.pool)
        .await
        .map_err(|e| PersistenceError::ReadFailed(e.to_string()))?;

        Ok(row.map(|(data,)| data))
    }

    async fn remove(&self, key: &QueryKey) -> Result<(), PersistenceError> {
        let hash = key.hash_hex();
        sqlx::query("DELETE FROM query_cache WHERE key_hash = ?")
            .bind(&hash)
            .execute(&self.pool)
            .await
            .map_err(|e| PersistenceError::RemoveFailed(e.to_string()))?;
        Ok(())
    }

    async fn clear(&self) -> Result<(), PersistenceError> {
        sqlx::query("DELETE FROM query_cache")
            .execute(&self.pool)
            .await
            .map_err(|e| PersistenceError::ClearFailed(e.to_string()))?;
        Ok(())
    }
}
```

## Registering a Persister

Attach a persister to your `QueryClient`:

```rust
let persister = Arc::new(FilePersister::new("./cache"));

let client = QueryClient::builder()
    .persister(persister.clone())
    .build();
```

The client will automatically:

- Persist data after successful fetches
- Restore data on startup for matching keys
- Remove data when queries are removed from cache

## Persistence Options

Control persistence behavior per-query:

```rust
let query = use_query(
    cx,
    QueryKey::from("user-settings"),
    |_| async move { load_settings().await },
    QueryOptions {
        persist: true,           // Enable persistence for this query
        persist_ttl: Some(Duration::from_secs(86400)), // Expire after 24 hours
        ..Default::default()
    },
);
```

<Callout type="info">
  Persistence is opt-in. By default, no persister is configured and query data lives only in memory.
</Callout>

<Callout type="warning">
  Avoid persisting sensitive data (tokens, passwords, personal information) without encryption.
  Implement an encrypted persister if your cache contains private data.
</Callout>
